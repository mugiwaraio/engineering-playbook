-- ============================================================
-- PostgreSQL 开发 / 测试项目初始化模板
--
-- 权限设计：
--
-- Database:
--   owner = postgres / 管理员账号
--
-- Schema:
--   app owner = llm_gatecheck_user
--
-- llm_gatecheck_user 可以：
--   CREATE / ALTER / DROP TABLE
--   CREATE / DROP INDEX
--   SELECT / INSERT / UPDATE / DELETE
--   TRUNCATE
--   SEQUENCE
--   VIEW
--   FUNCTION
--   TYPE
--   等项目开发操作
--
-- llm_gatecheck_user 不拥有：
--   SUPERUSER
--   CREATEDB
--   CREATEROLE
--   REPLICATION
--
-- 使用 postgres / 管理员账号执行
-- ============================================================


-- ============================================================
-- 1. 创建开发用户
-- ============================================================

CREATE USER llm_gatecheck_user
WITH
    PASSWORD 'xxx'
    NOSUPERUSER
    NOCREATEDB
    NOCREATEROLE
    NOREPLICATION;

-- ============================================================
-- 2. 创建项目数据库
--
-- 注意：
-- 不再指定 OWNER = llm_gatecheck_user
--
-- 数据库默认由当前管理员账号持有
-- ============================================================

CREATE DATABASE llm_gatecheck
WITH
    ENCODING = 'UTF8'
    LC_COLLATE = 'C.UTF-8'
    LC_CTYPE = 'C.UTF-8'
    TEMPLATE = template0;


-- ============================================================
-- 3. 收紧默认 PUBLIC 数据库权限
-- ============================================================

REVOKE ALL PRIVILEGES
ON DATABASE llm_gatecheck
FROM PUBLIC;


-- ============================================================
-- 4. 给开发账号数据库级完整权限
--
-- PostgreSQL DATABASE 级的 ALL 包括：
-- CREATE
-- CONNECT
-- TEMPORARY
-- ============================================================

GRANT ALL PRIVILEGES
ON DATABASE llm_gatecheck
TO llm_gatecheck_user;


-- ============================================================
-- 5. 切换项目数据库
--
-- 这是 psql 命令。
--
-- DBeaver / DataGrip / pgAdmin：
-- 手动切换到 llm_gatecheck 数据库后再执行后面的 SQL。
-- ============================================================

\c llm_gatecheck


-- ============================================================
-- 6. 禁止普通用户在 public schema 创建对象
-- ============================================================

REVOKE CREATE
ON SCHEMA public
FROM PUBLIC;


-- ============================================================
-- 7. 切换成项目开发用户
--
-- 关键：
-- app schema 由开发用户自己创建，
-- 因此 OWNER 自动就是 llm_gatecheck_user。
-- ============================================================

SET ROLE llm_gatecheck_user;


-- ============================================================
-- 8. 创建业务 Schema
-- ============================================================

CREATE SCHEMA app;


-- ============================================================
-- 9. 恢复管理员角色
-- ============================================================

RESET ROLE;


-- ============================================================
-- 10. 设置默认 Schema
-- ============================================================

ALTER ROLE llm_gatecheck_user
IN DATABASE llm_gatecheck
SET search_path TO app;


-- ============================================================
-- 11. 设置开发用户时区
-- ============================================================

ALTER ROLE llm_gatecheck_user
IN DATABASE llm_gatecheck
SET timezone TO 'UTC';


-- ============================================================
-- 12. 设置数据库默认时区
-- ============================================================

ALTER DATABASE llm_gatecheck
SET timezone TO 'UTC';


-- ============================================================
-- 初始化完成
-- ============================================================
